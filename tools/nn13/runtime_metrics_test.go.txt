package onnx

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"strings"
	"testing"

	nn13trace "github.com/itsmontoya/neuralnetwork/tools/nn13/trace"
)

func TestNN13PayloadMetrics(t *testing.T) {
	data, err := os.ReadFile("testdata/inference/provenance.json")
	if err != nil {
		t.Fatal(err)
	}

	var suite struct {
		Cases []conformanceCase `json:"cases"`
	}
	if err = json.Unmarshal(data, &suite); err != nil {
		t.Fatal(err)
	}

	for _, fixture := range suite.Cases {
		if !strings.HasPrefix(fixture.Name, "mlp_") && !strings.HasPrefix(fixture.Name, "shape_") {
			continue
		}

		t.Run(fixture.Name, func(t *testing.T) {
			modelData, readErr := fixture.Model.read()
			if readErr != nil {
				t.Fatal(readErr)
			}

			inputs, inputErr := fixture.readInputs()
			if inputErr != nil {
				t.Fatal(inputErr)
			}

			ctx := context.Background()
			nn13trace.Begin()
			model, loadErr := LoadBytes(ctx, modelData, Options{})
			if loadErr != nil {
				t.Fatal(loadErr)
			}

			if loadErr = model.Close(); loadErr != nil {
				t.Fatal(loadErr)
			}

			load := nn13trace.End()
			model, loadErr = LoadBytes(ctx, modelData, Options{})
			if loadErr != nil {
				t.Fatal(loadErr)
			}

			nn13trace.Begin(model, inputs)
			session, sessionErr := model.NewSession()
			if sessionErr != nil {
				t.Fatal(sessionErr)
			}

			outputs, runErr := session.Run(ctx, inputs, RunOptions{})
			if runErr != nil {
				t.Fatal(runErr)
			}

			if sessionErr = session.Close(); sessionErr != nil {
				t.Fatal(sessionErr)
			}

			first := nn13trace.End()
			if runErr = fixture.compareOutputs(outputs); runErr != nil {
				t.Fatal(runErr)
			}

			session, sessionErr = model.NewSession()
			if sessionErr != nil {
				t.Fatal(sessionErr)
			}

			retained, retainedErr := session.Run(ctx, inputs, RunOptions{})
			if retainedErr != nil {
				t.Fatal(retainedErr)
			}

			nn13trace.Begin(model, inputs, retained)
			outputs, runErr = session.Run(ctx, inputs, RunOptions{})
			if runErr != nil {
				t.Fatal(runErr)
			}

			warm := nn13trace.End()
			if sessionErr = session.Close(); sessionErr != nil {
				t.Fatal(sessionErr)
			}

			if loadErr = model.Close(); loadErr != nil {
				t.Fatal(loadErr)
			}

			if runErr = fixture.compareOutputs(outputs); runErr != nil {
				t.Fatal(runErr)
			}

			if retainedErr = fixture.compareOutputs(retained); retainedErr != nil {
				t.Fatal(retainedErr)
			}

			for phase, snapshot := range map[string]nn13trace.Snapshot{"Load": load, "First": first, "Warm": warm} {
				record := struct {
					Case     string             `json:"case"`
					Phase    string             `json:"phase"`
					Snapshot nn13trace.Snapshot `json:"metrics"`
				}{fixture.Name, phase, snapshot}
				encoded, encodeErr := json.Marshal(record)
				if encodeErr != nil {
					t.Fatal(encodeErr)
				}

				fmt.Printf("NN13METRIC %s\n", encoded)
			}
		})
	}
}
